## 必做 A 参考答案

### A1. import_lab.py

```python
def ping() -> str:
    return "pong"


def main() -> None:
    print(f"__name__ = {__name__!r}")
    print(ping())


if __name__ == "__main__":
    main()
```

**讲评**：`python import_lab.py` → `__name__ == '__main__'`；被 import 时不执行 `main()`。

### B1. safe_divide

```python
def safe_divide(a: float, b: float) -> float:
    if b == 0:
        raise ValueError("除数不能为 0")
    return a / b


def demo_safe_divide() -> None:
    for a, b in [(10, 2), (1, 0)]:
        try:
            result = safe_divide(a, b)
        except ValueError as exc:
            print(f"失败: {exc}")
        else:
            print(f"成功: {result}")
```

### B2. RateLimitError

```python
from sparktech import SparkTechError


class RateLimitError(SparkTechError):
    def __init__(self, message: str, *, retry_after: int) -> None:
        super().__init__(message, code="RATE_LIMIT")
        self.retry_after = retry_after


# main 中
try:
    raise RateLimitError("API 限流", retry_after=30)
except SparkTechError as exc:
    print(exc.code, getattr(exc, "retry_after", None))
```

### B3. read_lines_count

```python
from pathlib import Path

from sparktech import DataLoadError


def read_lines_count(path: Path) -> int:
    if not path.is_file():
        raise DataLoadError(f"文件不存在: {path}", path=str(path))
    f = None
    try:
        f = path.open(encoding="utf-8")
        return sum(1 for _ in f)
    finally:
        if f is not None:
            f.close()
```

**讲评**：更 Pythonic 的写法是 `with path.open() as f:`，`finally` 版用于理解语义。
