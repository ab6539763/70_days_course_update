### A1. memoize

```python
import functools
from typing import Any, Callable, TypeVar

F = TypeVar("F", bound=Callable[..., Any])

def memoize(func: F) -> F:
    cache: dict[tuple[Any, ...], Any] = {}

    @functools.wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        key = (args, tuple(sorted(kwargs.items())))
        if key not in cache:
            print("计算中…")
            cache[key] = func(*args, **kwargs)
        return cache[key]

    return wrapper  # type: ignore[return-value]


@memoize
def expensive_add(a: int, b: int) -> int:
    return a + b


if __name__ == "__main__":
    print(expensive_add(1, 2))  # 计算中… 3
    print(expensive_add(1, 2))  # 3（无「计算中」）
```

**讲评**：缓存 key 必须可哈希；可变默认参数不要放在装饰器外层。

### A2. retry_lab

```python
_attempts = 0

@retry(max_attempts=4, delay=0.1, exceptions=(ZeroDivisionError,))
def unstable_divide(a: float, b: float) -> float:
    global _attempts
    _attempts += 1
    if _attempts < 3 and b == 0:
        raise ZeroDivisionError("模拟不稳定")
    return a / b

print(unstable_divide(10, 2))  # 5.0
```

### B1. iter_errors

```python
from collections.abc import Generator
from pathlib import Path

def iter_errors(path: str) -> Generator[str, None, None]:
    with Path(path).open(encoding="utf-8") as f:
        for line in f:
            if "ERROR" in line:
                yield line.rstrip("\n")
```

### B2. batched

```python
from collections.abc import Generator, Iterable
from typing import TypeVar

T = TypeVar("T")

def batched(iterable: Iterable[T], size: int) -> Generator[list[T], None, None]:
    batch: list[T] = []
    for item in iterable:
        batch.append(item)
        if len(batch) >= size:
            yield batch
            batch = []
    if batch:
        yield batch
```
