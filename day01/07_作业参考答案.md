## B2 参考答案

`personal_info_card_hw.py` 核心逻辑参考：

```python
# -*- coding: utf-8 -*-
"""B2 参考答案：部门列表与序号选择。"""

from __future__ import annotations

from constants import (
    BOOL_DISPLAY_FALSE,
    BOOL_DISPLAY_TRUE,
    CARD_SUBTITLE,
    CARD_WIDTH,
    COMPANY_NAME,
    DEFAULT_DEPARTMENT,
    DEPARTMENTS,
    LINE_CHAR_MAIN,
    LINE_CHAR_SUB,
    PROMPT_EMPLOYEE_ID,
    PROMPT_NAME,
    PROMPT_NEED_DORM,
    PROMPT_ONBOARD_DATE,
    PROMPT_PHONE,
    SAMPLE_EMPLOYEE_ID,
)
from datetime import datetime


def print_department_menu() -> None:
    """打印部门序号菜单，供 US 扩展使用。"""
    print("可选部门：")
    for index, dept_name in enumerate(DEPARTMENTS, start=1):
        print(f"  {index}. {dept_name}")
    print(f"  直接回车默认：{DEFAULT_DEPARTMENT}")


def resolve_department(raw: str) -> str:
    """
    解析部门输入：序号 / 全名 / 空 / 非法。

    Returns:
        最终部门名称
    """
    text = raw.strip()
    if not text:
        return DEFAULT_DEPARTMENT

    # 纯数字则尝试按序号解析
    if text.isdigit():
        idx = int(text)
        if 1 <= idx <= len(DEPARTMENTS):
            return DEPARTMENTS[idx - 1]
        print(f"[警告] 序号 {idx} 无效，已使用默认部门。")
        return DEFAULT_DEPARTMENT

    # 全名精确匹配（v1 大小写敏感；Day 2 可用 .lower() 放宽）
    if text in DEPARTMENTS:
        return text

    print(f"[警告] 未识别部门「{text}」，已使用默认部门。")
    return DEFAULT_DEPARTMENT


def parse_need_dorm(raw: str) -> bool:
    return raw.strip().lower() in ("y", "yes", "是", "1", "要")


def format_bool_display(value: bool) -> str:
    return BOOL_DISPLAY_TRUE if value else BOOL_DISPLAY_FALSE


def main() -> None:
    print_department_menu()
    print()

    name = input("请输入姓名：").strip() or "（未填写）"
    employee_id = input(PROMPT_EMPLOYEE_ID).strip() or SAMPLE_EMPLOYEE_ID
    department = resolve_department(input("请输入部门序号或名称："))
    onboard_date = input(PROMPT_ONBOARD_DATE).strip() or datetime.now().strftime("%Y-%m-%d")
    phone = input(PROMPT_PHONE).strip() or "（未填写）"
    need_dorm = parse_need_dorm(input(PROMPT_NEED_DORM))
    generated_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    main_line = LINE_CHAR_MAIN * CARD_WIDTH
    sub_line = LINE_CHAR_SUB * CARD_WIDTH

    print()
    print(main_line)
    print(f"{COMPANY_NAME} · {CARD_SUBTITLE}".center(CARD_WIDTH))
    print(main_line)
    print(f"  姓    名：{name}")
    print(f"  工    号：{employee_id}")
    print(f"  部    门：{department}")
    print(f"  入职日期：{onboard_date}")
    print(f"  联系电话：{phone}")
    print(f"  是否住宿：{format_bool_display(need_dorm)}")
    print(sub_line)
    print(f"  生成时间：{generated_at}")
    print(main_line)


if __name__ == "__main__":
    main()
```

**常见失分**：

- 序号从 0 开始（应与 `enumerate(..., start=1)` 一致）  
- 非法输入未给警告  
- 复制课堂版未实现菜单

## B3 参考答案

`type_playground.py`：

```python
# -*- coding: utf-8 -*-
"""B3 参考答案。"""

a, b = 10, 3

print("a/b =", a / b, type(a / b))
# / 是真除法，10/3=3.333...，结果为 float

print("a//b =", a // b, type(a // b))
# // 是地板除，向负无穷取整；正数时即整除商

s = "100"
num = int(s) + 1
print("int(s)+1 =", num)
# input 与 JSON 数字字段常需 int() / float() 显式转换

flag = "false"
print('bool("false") =', bool(flag))
# 非空字符串在布尔上下文均为 True；只有 "" 为 False
```

## 答疑摘要（往届高频三问）

**Q1：`pip` 不是内部命令？**  
A：用 `python3 -m pip` 代替 `pip`。

**Q2：VS Code 选错解释器？**  
A：`Ctrl+Shift+P` → `Python: Select Interpreter` → 选 3.10+。

**Q3：Git push 要密码？**  
A：推荐配置 SSH key 或使用 GitHub Personal Access Token。
