## 必做 A 参考答案

### A1–A2. mini_http.py

```python
import json

import requests

TIMEOUT = 10


def demo_get_uuid() -> None:
    url = "https://httpbin.org/uuid"
    response = requests.get(url, timeout=TIMEOUT)
    print(f"status_code={response.status_code}")
    print(f"uuid={response.json()['uuid']}")


def demo_llm_shaped_post() -> None:
    url = "https://httpbin.org/post"
    payload = {
        "model": "deepseek-chat",
        "messages": [{"role": "user", "content": "作业 POST 演示"}],
    }
    response = requests.post(url, json=payload, timeout=TIMEOUT)
    print(f"status_code={response.status_code}")
    print(json.dumps(response.json().get("json"), ensure_ascii=False, indent=2))


if __name__ == "__main__":
    demo_get_uuid()
    demo_llm_shaped_post()
```

**讲评**：与 DeepSeek 的相同点是 **POST + JSON body + `messages` 列表**；差异是 URL、鉴权头与响应解析路径。
