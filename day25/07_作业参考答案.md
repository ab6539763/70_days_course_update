```python
from langchain_core.prompts import ChatPromptTemplate, PromptTemplate

ticket_tpl = PromptTemplate(
    input_variables=["ticket_id", "tone"],
    template="请用{tone}语气，总结工单 {ticket_id} 的处理要点。",
)

chat_tpl = ChatPromptTemplate.from_messages([
    ("system", "你是{company}的智能客服助手。"),
    ("human", "{user_input}"),
])
print(ticket_tpl.format(ticket_id="TK-001", tone="专业"))
print(chat_tpl.format_messages(company="星火智服", user_input="你好"))
```

加载逻辑：

```python
import json
from pathlib import Path

def load_system_prompt() -> str:
    profile = os.getenv("SPARKTECH_SYSTEM_PROFILE", "default")
    data = json.loads(Path("system_prompts.json").read_text(encoding="utf-8"))
    return data.get(profile, data["default"])
```
